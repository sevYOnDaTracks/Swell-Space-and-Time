cmake_minimum_required(VERSION 3.22)
project(SwellSpaceTime VERSION 0.1.1 LANGUAGES C CXX)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
set(SPACE_FORMATS VST3 Standalone)
if(APPLE)
  list(APPEND SPACE_FORMATS AU)
endif()
juce_add_plugin(SwellSpaceTime COMPANY_NAME "Swell-audio" BUNDLE_ID "fr.swell-audio.spacetime" PLUGIN_MANUFACTURER_CODE Swel PLUGIN_CODE Sspt FORMATS ${SPACE_FORMATS} PRODUCT_NAME "Swell Space and Time" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
juce_add_binary_data(SpaceAssets SOURCES swell-mark.png)
target_sources(SwellSpaceTime PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp)
target_compile_definitions(SwellSpaceTime PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(SwellSpaceTime PRIVATE SpaceAssets juce::juce_audio_utils juce::juce_dsp PUBLIC juce::juce_recommended_config_flags)
enable_testing()
juce_add_console_app(SpaceTest PRODUCT_NAME "Space Audio Tests")
target_sources(SpaceTest PRIVATE Tests/AudioTest.cpp)
target_include_directories(SpaceTest PRIVATE "${juce_SOURCE_DIR}/modules")
target_compile_definitions(SpaceTest PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(SpaceTest PRIVATE SwellSpaceTime)
add_test(NAME SpaceAudio COMMAND SpaceTest)
juce_add_console_app(SpaceDemo PRODUCT_NAME "Space Demo Renderer")
target_sources(SpaceDemo PRIVATE Tools/RenderDemo.cpp)
target_include_directories(SpaceDemo PRIVATE "${juce_SOURCE_DIR}/modules")
target_compile_definitions(SpaceDemo PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(SpaceDemo PRIVATE SwellSpaceTime)
juce_add_console_app(PingTest PRODUCT_NAME "Ping Pong Tests")
target_sources(PingTest PRIVATE Tests/PingTest.cpp)
target_include_directories(PingTest PRIVATE "${juce_SOURCE_DIR}/modules")
target_compile_definitions(PingTest PRIVATE JUCE_GLOBAL_MODULE_SETTINGS_INCLUDED=1)
target_link_libraries(PingTest PRIVATE SwellSpaceTime)
add_test(NAME PingPong COMMAND PingTest)
